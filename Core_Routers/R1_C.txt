hostname R1C
!
no ip domain-lookup
ip cef
!
segment-routing mpls
 !
 set-attributes
  address-family ipv4
   sr-label-preferred
  exit-address-family
 !
 connected-prefix-sid-map
  address-family ipv4
   10.255.1.1/32 index 10 range 1
  exit-address-family
!
! ================================================================
! IS-IS process must exist with segment-routing mpls enabled BEFORE
! any interface attempts "isis prefix-sid index N" — otherwise the
! parser rejects the prefix-sid line on a fresh boot.
! ================================================================
router isis CORE
 net 49.0001.0102.5500.1001.00
 is-type level-2-only
 metric-style wide
 passive-interface Loopback0
 fast-reroute per-prefix level-2 all
 segment-routing mpls
!
interface Loopback0
 ip address 10.255.1.1 255.255.255.255
 ip router isis CORE
!
interface GigabitEthernet0/0/0
 description TO-R2C
 ip address 10.0.12.1 255.255.255.252
 ip router isis CORE
 isis network point-to-point
 no shutdown
!
interface GigabitEthernet0/0/1
 description TO-R3C
 ip address 10.0.13.1 255.255.255.252
 ip router isis CORE
 isis network point-to-point
 no shutdown
!
interface GigabitEthernet0/0/2
 description TO-R1E
 ip address 10.0.101.1 255.255.255.252
 ip router isis CORE
 isis network point-to-point
 no shutdown
!
! (We will make R1C one of the Route Reflectors)
router bgp 65000
 bgp log-neighbor-changes
 neighbor 10.255.11.11 remote-as 65000
 neighbor 10.255.11.11 update-source Loopback0
 neighbor 10.255.12.12 remote-as 65000
 neighbor 10.255.12.12 update-source Loopback0
 neighbor 10.255.13.13 remote-as 65000
 neighbor 10.255.13.13 update-source Loopback0
 !
 address-family vpnv4
  no bgp default route-target filter
  neighbor 10.255.11.11 activate
  neighbor 10.255.11.11 send-community extended
  neighbor 10.255.11.11 route-reflector-client
  neighbor 10.255.12.12 activate
  neighbor 10.255.12.12 send-community extended
  neighbor 10.255.12.12 route-reflector-client
  neighbor 10.255.13.13 activate
  neighbor 10.255.13.13 send-community extended
  neighbor 10.255.13.13 route-reflector-client
 exit-address-family
!
end
