! ======================================
! Hostname and basic settings
! ======================================
hostname SW_AGR1
!
no ip domain-lookup
ip routing
ip cef
!
! ======================================
! VRF Definitions
! ======================================
ip vrf CLIENT1
rd 65021:1
route-target export 65021:1
route-target import 65021:1
!
ip vrf CLIENT2
rd 65021:2
route-target export 65021:2
route-target import 65021:2
!
! ======================================
! Interfaces to PE Router (R1E)
! ======================================
interface GigabitEthernet1/0/1
description TO-R1E
ip address 172.16.1.1 255.255.255.252
no shutdown
!
! ======================================
! Client Access Interfaces
! Each client gets its own VRF
! ======================================
interface GigabitEthernet1/0/10
 description CLIENT1_SITE
 ip vrf forwarding CLIENT1
 ip address 172.16.11.2 255.255.255.252
 no shutdown
!
interface GigabitEthernet1/0/11
 description CLIENT2_SITE
 ip vrf forwarding CLIENT2
 ip address 172.16.12.2 255.255.255.252
 no shutdown
!
! ======================================
! eBGP to PE Router
! Each VRF peers with PE
! ======================================
router bgp 65021
 bgp log-neighbor-changes
!
 ! CLIENT1 VRF
address-family ipv4 vrf CLIENT1
neighbor 172.16.1.1 remote-as 65000
neighbor 172.16.1.1 activate
exit-address-family
!
 ! CLIENT2 VRF
address-family ipv4 vrf CLIENT2
neighbor 172.16.1.1 remote-as 65000
neighbor 172.16.1.1 activate
exit-address-family
!
! ======================================
! Summary
! VRF-Lite + eBGP to PE is ready
! No routing leakage yet; clients isolated
! ======================================
end
