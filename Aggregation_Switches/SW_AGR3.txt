! ================================================================
! SW_AGR3 — Catalyst 3750-E — Site 3 DC Aggregation Switch
! Connected to R3E via GigabitEthernet1/0/1 (802.1Q trunk)
! CE AS: 65023 | PE AS: 65000
!
! This switch is the DC hub:
!   1. DC_CENTRAL — Slican IP-PBX + management (10.100.x.x)
!   2. DC_PUB     — Supermicro public/general compute (203.0.113.0/24)
!
! Legacy CLIENT5 tenant has been removed; client tenants now live
! on SW_AGR1 (CLIENT1-20 + SHARED_101) and SW_AGR2 (CLIENT21-40 +
! SHARED_102). DC reachability is governed on R3E via 65000:1000
! (TO_DC_PUB) and 65000:2000 (TO_DC_CENTRAL) community RTs.
! ================================================================
!
hostname SW_AGR3
!
no ip domain-lookup
ip routing
ip cef
!
! ================================================================
! VTP / Spanning-Tree Hardening
! ================================================================
vtp mode transparent
spanning-tree mode rapid-pvst
spanning-tree extend system-id
!
! ================================================================
! VRF Definitions (IOS classic "ip vrf" for 3750-E)
! ================================================================
ip vrf DC_CENTRAL
 rd 65023:100
 route-target export 65023:100
 route-target import 65023:100
!
ip vrf DC_PUB
 rd 65023:99
 route-target export 65023:99
 route-target import 65023:99
!
! ================================================================
! VLANs
! ================================================================
vlan 200
 name DC_CENTRAL_PE_LINK
!
vlan 201
 name DC_PUB_PE_LINK
!
vlan 500
 name DC_SERVERS
!
vlan 510
 name DC_MGMT
!
vlan 600
 name DC_PUBLIC_VMS
!
! ================================================================
! Uplink to PE Router R3E — 802.1Q Trunk
! Carries VLANs 200, 201 toward PE sub-interfaces
! ================================================================
interface GigabitEthernet1/0/1
 description TRUNK-TO-R3E-Gi0/0/0
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 200,201
 no shutdown
!
! ================================================================
! SVI — PE-facing (one per VRF, matching PE sub-if VLAN)
! ================================================================
interface Vlan200
 description DC-CENTRAL-PE-LINK
 ip vrf forwarding DC_CENTRAL
 ip address 172.16.100.2 255.255.255.252
 no shutdown
!
interface Vlan201
 description DC-PUB-PE-LINK
 ip vrf forwarding DC_PUB
 ip address 172.16.199.2 255.255.255.252
 no shutdown
!
! ================================================================
! DC_PUB — Public VMs Network (Supermicro general compute)
! ================================================================
interface Vlan600
 description PUBLIC-VMS-GATEWAY
 ip vrf forwarding DC_PUB
 ip address 203.0.113.1 255.255.255.0
 no shutdown
!
! ================================================================
! DC_CENTRAL — Server VLAN (Slican PBX + management)
! ================================================================
interface Vlan500
 description DC-SERVERS-GATEWAY
 ip vrf forwarding DC_CENTRAL
 ip address 10.100.0.1 255.255.255.0
 no shutdown
!
interface Vlan510
 description DC-MGMT-GATEWAY
 ip vrf forwarding DC_CENTRAL
 ip address 10.100.10.1 255.255.255.0
 no shutdown
!
! ================================================================
! DC Server access ports
! ================================================================
interface GigabitEthernet1/0/20
 description SUPERMICRO-SERVER-DATA
 switchport access vlan 500
 switchport mode access
 spanning-tree portfast
 no shutdown
!
interface GigabitEthernet1/0/21
 description SUPERMICRO-SERVER-MGMT
 switchport access vlan 510
 switchport mode access
 spanning-tree portfast
 no shutdown
!
interface GigabitEthernet1/0/22
 description SLICAN-VOIP-SERVER
 switchport access vlan 500
 switchport mode access
 spanning-tree portfast
 no shutdown
!
interface GigabitEthernet1/0/23
 description PUBLIC-VMS-SERVER
 switchport access vlan 600
 switchport mode access
 spanning-tree portfast
 no shutdown
!
! ================================================================
! OSPF VRF-Lite — DC VRFs feed default-route to their downstream
! servers and redistribute server LANs up into eBGP.
! ================================================================
router ospf 100 vrf DC_CENTRAL
 default-information originate always
 redistribute bgp 65023 subnets
 network 10.100.0.0 0.0.0.255 area 0
 network 10.100.10.0 0.0.0.255 area 0
!
router ospf 199 vrf DC_PUB
 default-information originate always
 redistribute bgp 65023 subnets
 network 203.0.113.0 0.0.0.255 area 0
!
! ================================================================
! eBGP to PE Router R3E (AS 65000)
! ================================================================
router bgp 65023
 bgp router-id 192.168.255.3
 bgp log-neighbor-changes
 !
 address-family ipv4 vrf DC_CENTRAL
  redistribute ospf 100
  neighbor 172.16.100.1 remote-as 65000
  neighbor 172.16.100.1 activate
 exit-address-family
 !
 address-family ipv4 vrf DC_PUB
  redistribute ospf 199
  neighbor 172.16.199.1 remote-as 65000
  neighbor 172.16.199.1 activate
 exit-address-family
!
! ================================================================
! Management & Security Hardening
! ================================================================
no ip http server
no ip http secure-server
!
line con 0
 logging synchronous
 exec-timeout 30 0
line vty 0 15
 login local
 transport input ssh
!
end
