hostname R3E
!
! ================================================================
! VRF Definitions — MUST appear before interfaces that reference them
! On IOS-XE, route-target MUST be inside address-family ipv4
! ================================================================
!
vrf definition CLIENT5
 rd 65000:5
 !
 address-family ipv4
  route-target export 65000:5
  route-target import 65000:5
  route-target import 65000:100
 exit-address-family
!
vrf definition DC_CENTRAL
 rd 65000:100
 !
 address-family ipv4
  route-target export 65000:100
  route-target import 65000:100
  route-target import 65000:1
  route-target import 65000:2
  route-target import 65000:3
  route-target import 65000:4
  route-target import 65000:5
 exit-address-family
!
! ================================================================
! Loopback — Router-ID, LDP-ID, BGP update-source
! ================================================================
interface Loopback0
 ip address 10.255.13.13 255.255.255.255
!
! ================================================================
! WAN Uplink to Core P Router R3C
! ================================================================
interface GigabitEthernet0/0/2
 description TO-R3C
 ip address 10.0.103.2 255.255.255.252
 mpls ip
 no shutdown
!
! ================================================================
! Trunk to Aggregation Switch SW_AGR3
! Carries VLAN 105 (CLIENT5) and VLAN 200 (DC_CENTRAL)
! ================================================================
interface GigabitEthernet0/0/0
 description TRUNK-TO-SW_AGR3
 no ip address
 no shutdown
!
interface GigabitEthernet0/0/0.105
 description CLIENT5
 encapsulation dot1Q 105
 vrf forwarding CLIENT5
 ip address 172.16.5.1 255.255.255.252
!
interface GigabitEthernet0/0/0.200
 description DC_CENTRAL
 encapsulation dot1Q 200
 vrf forwarding DC_CENTRAL
 ip address 172.16.100.1 255.255.255.252
!
! ================================================================
! OSPF — Backbone IGP (transport only)
! ================================================================
router ospf 1
 router-id 10.255.13.13
 network 10.255.13.13 0.0.0.0 area 0
 network 10.0.103.0 0.0.0.3 area 0
!
! ================================================================
! MPLS LDP
! ================================================================
mpls label protocol ldp
mpls ldp router-id Loopback0 force
!
! ================================================================
! MP-BGP — iBGP VPNv4 full mesh + per-VRF eBGP to CE
! ================================================================
router bgp 65000
 bgp log-neighbor-changes
 !
 neighbor 10.255.11.11 remote-as 65000
 neighbor 10.255.11.11 update-source Loopback0
 !
 neighbor 10.255.12.12 remote-as 65000
 neighbor 10.255.12.12 update-source Loopback0
 !
 address-family vpnv4
  neighbor 10.255.11.11 activate
  neighbor 10.255.11.11 send-community extended
  neighbor 10.255.12.12 activate
  neighbor 10.255.12.12 send-community extended
 exit-address-family
 !
 address-family ipv4 vrf CLIENT5
  neighbor 172.16.5.2 remote-as 65023
  neighbor 172.16.5.2 activate
 exit-address-family
 !
 address-family ipv4 vrf DC_CENTRAL
  neighbor 172.16.100.2 remote-as 65023
  neighbor 172.16.100.2 activate
 exit-address-family
!
end
